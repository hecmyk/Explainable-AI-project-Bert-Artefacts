from pathlib import Path
import shutil

ROOT = Path("data/FinRAGBench-V")
PDFS = ROOT / "pdfs"
PROCESSED = ROOT / "processed"

REQUIRED = ["markdown.md", "doc_ocr.json", "stats.json"]  # marqueurs de "traité"

kept, removed = 0, 0
for d in PROCESSED.iterdir():
    if not d.is_dir():
        continue
    if all((d / f).exists() for f in REQUIRED):
        pdf = PDFS / f"{d.name}.pdf"
        if pdf.exists():
            shutil.copy2(pdf, d / pdf.name)  # ou shutil.move si tu veux déplacer
        kept += 1
    else:
        shutil.rmtree(d)
        removed += 1

print(f"kept: {kept}, removed: {removed}")
