from collections import Counter

FORMS = [("valid", r"^(s\d+|s_toc)$"), ("atom_id", r"^p\d+_e\d+$"), ("hybrid", r"^s\d+_e\d+$")]
counts, bad_per_query, examples = Counter(), [], Counter()

for calls in df["get_section_calls"]:
    bad = 0
    for cid in (calls if isinstance(calls, list) else []):
        form = next((n for n, p in FORMS if cid and re.match(p, str(cid))), "other")
        counts[form] += 1
        if form != "valid":
            bad += 1
            examples[str(cid)] += 1
    bad_per_query.append(bad)

df["n_bad_ids"] = bad_per_query
total = sum(counts.values())
for form, n in counts.most_common():
    print(f"{form:10s} {n:5d}  ({n / total:.1%})")
print(f"\nqueries avec >=1 id invalide : {(df['n_bad_ids'] > 0).mean():.1%}")
print("exemples:", examples.most_common(10))
